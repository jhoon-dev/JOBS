cmake_minimum_required(VERSION 3.24)
project(JOBS VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/utf-8 /W4)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Widgets Concurrent)
add_library(jobs_domain src/domain/Project.cpp)
target_include_directories(jobs_domain PUBLIC src)
add_library(jobs_application src/application/Document.cpp)
add_library(jobs_calculation src/calculation/Scheduler.cpp)
target_link_libraries(jobs_calculation PUBLIC jobs_domain)
target_link_libraries(jobs_application PUBLIC jobs_domain jobs_calculation)
add_library(jobs_persistence src/persistence/JsonProjectStore.cpp)
target_link_libraries(jobs_persistence PUBLIC jobs_application Qt6::Core)
add_library(jobs_ui src/ui/MainWindow.cpp)
target_link_libraries(jobs_ui PUBLIC jobs_application Qt6::Widgets Qt6::Concurrent)
add_executable(JOBS WIN32 MACOSX_BUNDLE src/main.cpp)
target_link_libraries(JOBS PRIVATE jobs_ui jobs_persistence)
set_target_properties(JOBS PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER com.toolmoa.jobs)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(jobs_schedule_tests tests/SchedulerTests.cpp)
  target_link_libraries(jobs_schedule_tests PRIVATE jobs_calculation)
  add_test(NAME schedule_contract COMMAND jobs_schedule_tests)
  add_executable(jobs_file_tests tests/FileLifecycleTests.cpp)
  target_link_libraries(jobs_file_tests PRIVATE jobs_persistence)
  add_test(NAME file_lifecycle COMMAND jobs_file_tests)
  add_executable(jobs_ui_tests tests/UiLifecycleTests.cpp)
  target_link_libraries(jobs_ui_tests PRIVATE jobs_ui jobs_persistence Qt6::Test)
  add_test(NAME ui_lifecycle COMMAND jobs_ui_tests -platform offscreen)
endif()
install(TARGETS JOBS BUNDLE DESTINATION . RUNTIME DESTINATION bin)
